    using System.Collections.Generic;

public class ColorFilterSignalBehavior
    : ISignalBehavior
{
    private readonly
        SignalColor allowedColor;


    public ColorFilterSignalBehavior(
        SignalColor allowedColor)
    {
        this.allowedColor =
            allowedColor;
    }


    public IReadOnlyList<SignalOutput>
        Process(
            SignalColor inputColor,
            Direction enteredFrom,
            IReadOnlyList<Direction>
                connections)
    {
        if (inputColor !=
            allowedColor)
        {
            return
                System.Array.Empty<
                    SignalOutput>();
        }


        List<SignalOutput>
            outputs =
                new List<SignalOutput>();


        foreach (
            Direction direction
            in connections)
        {
            if (direction ==
                enteredFrom)
            {
                continue;
            }


            outputs.Add(
                new SignalOutput(
                    direction,
                    inputColor
                )
            );
        }


        return outputs;
    }
}
