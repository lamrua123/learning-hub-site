def build_answer_record(question, answer, status, earned_points):
    return {
        "prompt": question["prompt"],
        "topic": question["topic"],
        "answer": answer,
        "correct_answer": question["correct_answer"],
        "status": status,
        "earned_points": earned_points,
        "possible_points": question["points"]
    }

def submit_answer(session, question, raw_answer):
    answer = normalize_answer(raw_answer)

    if not is_valid_answer(answer):
        return "invalid"

    if answer == "skip":
        session["skipped_count"] += 1
        record = build_answer_record(question, answer, "skipped", 0)
        session["answer_history"].append(record)
        return "skipped"

    if answer == question["correct_answer"]:
        earned_points = question["points"]
        session["score"] += earned_points
        session["correct_count"] += 1
        status = "correct"
    else:
        earned_points = 0
        session["incorrect_count"] += 1
        status = "incorrect"

    record = build_answer_record(
        question,
        answer,
        status,
        earned_points
    )
    session["answer_history"].append(record)

    return status

def submit_answer(session, question, raw_answer):
    answer = normalize_answer(raw_answer)

    if not is_valid_answer(answer):
        return "invalid"

    if answer == "skip":
        session["skipped_count"] += 1
        record = build_answer_record(question, answer, "skipped", 0)
        session["answer_history"].append(record)
        return "skipped"

    if answer == question["correct_answer"]:
        earned_points = question["points"]
        session["score"] += earned_points
        session["correct_count"] += 1
        status = "correct"
    else:
        earned_points = 0
        session["incorrect_count"] += 1
        status = "incorrect"

    record = build_answer_record(
        question,
        answer,
        status,
        earned_points
    )
    session["answer_history"].append(record)

    return status

def run_quiz(questions, passing_percentage):
    session = build_session()

    for index in range(len(questions)):
        question = questions[index]
        show_question(question, index + 1, len(questions))

        while True:
            raw_answer = input("Answer a/b/c hoặc skip: ")
            status = submit_answer(session, question, raw_answer)

            if status == "invalid":
                print("Answer không hợp lệ, hãy thử lại")
                continue

            show_feedback(status, question)
            break

    report = build_report(session, questions, passing_percentage)
    return report

    • correct;
    • incorrect.

    So sánh mọi counter, score và history length.

    Bài 4 — Report invariants

    Viết function is_session_consistent(session, question_count) trả Boolean.

    • tổng counters bằng question count;
    • history length bằng question count;
    • score không âm;
    • gọi với state đúng và state cố tình sai.

    Bài 5 — Topic report

    Duyệt answer history records và đếm:

    • correct theo topic python;
    • incorrect theo topic math;
    • tổng earned/possible points;
    • không dùng comprehension.

    Bài 6 — Attempt tracker

    Record ba reports theo thứ tự 50%, 90%, 70%.

    • attempts là 3;
    • best là 90%;
    • last là report 70%;
    • chứng minh best khác last.

    Bài 7 — Review mode

    Viết presentation function duyệt history và in:

    • prompt;
    • answer người dùng;
    • correct answer;
    • status;
    • earned/possible points.

    Presentation không mutation record.

    Bài 8 — Empty quiz

    Chạy pipeline với questions rỗng.

    • session mới;
    • max score 0;
    • percentage None;
    • passed False;
    • tracker vẫn có thể record attempt theo rule bạn chọn;
    • giải thích design choice.

    40. Điều cần nhớ

    • List của dictionaries phù hợp cho nhiều records cùng cấu trúc.
    • Nested access nên được đọc từng bước và dùng biến trung gian khi cần.
    • Session giữ state một attempt; tracker giữ state qua nhiều attempts.
    • Mỗi attempt cần session và history mới.
    • Invalid answer phải dừng trước mọi mutation.
    • Correct, incorrect và skipped có state transitions khác nhau.
    • Mỗi answer hợp lệ tạo đúng một history record.
    • Maximum score là tổng points; percentage cần bảo vệ trường hợp max bằng 0.
    • None khác 0%: một bên là chưa có tỷ lệ, một bên là kết quả thật.
    • Best percentage chỉ tăng; last report luôn đổi theo attempt gần nhất.
    • Presentation không chấm điểm hoặc mutation session.
    • Nested loop hỏi lại invalid answer nhưng vẫn tiến qua từng question.
    • Report được build từ session đã kiểm tra nhất quán.
    • Scenario tests cần setup state độc lập và kiểm tra cả đường thành công/thất bại.
    • Pattern question/session/report/tracker chuyển được sang form, survey và automation checklist.
